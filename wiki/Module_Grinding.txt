-- Module:Grinding — builds grinding-style recipe tables (Name / Ingredients / Recipe [/ Desc]).
-- Clean-room implementation. Mirrors Module:Smelting but with a single input (no fuel); the
-- Recipe cell reuses our own {{Grid Grinding Table}} widget, which also renders an optional
-- chance byproduct. The generic builder p.build is shared with [[Module:Sawmill]] (which
-- requires this module and just passes a different column label and no experience).
-- head=1 opens the table + header, foot=1 closes it.

local itemlink = require('Module:ItemLink')

-- Local page link for a mod item, plain text for anything else. Uses [[Module:ItemLink]]'s
-- table (targetFor returns the bare name for a mod item, a Minecraft Wiki URL for a vanilla
-- one) instead of a title-existence check, which is an EXPENSIVE parser call: Fandom allows
-- only 100 of those per page and a long recipe table blows straight through the cap.
local function linkIfMod(v)
	return (itemlink.targetFor(v) == v) and ('[[' .. v .. ']]') or v
end

local p = {}

local function trim(s)
	if s == nil then return nil end
	s = mw.text.trim(s)
	if s == '' then return nil end
	return s
end

local function truthy(v)
	v = trim(v)
	return v ~= nil and v ~= '0'
end

-- "Name,Count" -> name, count (also honours an explicit OutputCount)
local function parseOutput(args)
	local out = trim(args.Output) or trim(args.output) or ''
	local count = trim(args.OutputCount) or trim(args.outputcount)
	local n, c = out:match('^(.-)%s*,%s*(%d+)$')
	if n then
		out = n
		count = count or c
	end
	return out, count
end

-- "Name,Chance" -> name, chance (also honours an explicit ByproductChance)
local function parseByproduct(args)
	local bp = trim(args.Byproduct) or trim(args.byproduct)
	if not bp then return nil, nil end
	local chance = trim(args.ByproductChance) or trim(args.byproductchance)
	local n, c = bp:match('^(.-)%s*,%s*([%d%.]+%%?)$')
	if n then
		bp = n
		chance = chance or c
	end
	return bp, chance
end

-- Ingredients column: explicit |ingredients=, else the input (linked if it has a wiki page).
local function ingredientsCell(args)
	local ing = trim(args.ingredients)
	if ing then return ing end
	-- split the ;-separated inputs, link those with a page, join the alternatives with "or"
	local parts = {}
	for part in (trim(args.Input) or trim(args.input) or ''):gmatch('[^;]+') do
		local v = trim(part)
		if v then
			parts[#parts + 1] = linkIfMod(v)
		end
	end
	local out = {}
	for i, item in ipairs(parts) do
		out[#out + 1] = '<span class="logi-recipes-ingredient">' .. item .. ((i < #parts) and ' or' or '') .. '</span>'
	end
	return table.concat(out)
end

-- Recipe column: our grinding widget (input -> output, optional byproduct + experience).
local function recipeCell(frame, args, outName, outCount, useXp)
	local g = { Output = outName }
	if outCount then g.OutputCount = outCount end
	local input = trim(args.Input) or trim(args.input)
	if input then g.Input = input end
	local bp, bpc = parseByproduct(args)
	if bp then
		g.Byproduct = bp
		if bpc then g.ByproductChance = bpc end
	end
	if useXp then
		local xp = trim(args.XP) or trim(args.xp)
		if xp then g.XP = xp end
	end
	local grid = frame:expandTemplate{ title = 'Grid Grinding Table', args = g }
	grid = grid:gsub('<templatestyles.-/>', '')
	return mw.text.trim(grid)
end

-- Generic builder shared by {{Grinding}} (Macerator) and {{Sawmill}}.
-- opts.label = the Recipe column header; opts.xp = whether to surface the XP param.
function p.build(frame, opts)
	local args = frame:getParent().args
	local outName, outCount = parseOutput(args)
	local showName = trim(args.showname) ~= '0'
	local showDesc = truthy(args.showdescription) or trim(args.description) ~= nil
	local head = truthy(args.head)
	local foot = truthy(args.foot)

	local out = {}

	if head then
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Crafting/styles.css' } }
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Grid Crafting Table/styles.css' } }
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Grid Grinding Table/styles.css' } }
		out[#out + 1] = '\n{| class="logi-recipes"'
		out[#out + 1] = '\n|- class="logi-recipes-head"'
		if showName then out[#out + 1] = '\n! Name' end
		out[#out + 1] = '\n! Ingredients\n! ' .. opts.label
		if showDesc then out[#out + 1] = '\n! Description' end
	end

	out[#out + 1] = '\n|-'
	if showName then
		local nm = trim(args.name) or ('[[' .. outName .. ']]')
		out[#out + 1] = '\n| class="logi-recipes-name" | ' .. nm
	end
	local ingredients = ingredientsCell(args)
	local notes = trim(args.notes)
	if notes then
		ingredients = ingredients .. '<div class="logi-recipes-note">' .. notes .. '</div>'
	end
	out[#out + 1] = '\n| class="logi-recipes-ingredients" | ' .. ingredients
	out[#out + 1] = '\n| class="logi-recipes-recipe" |\n' .. recipeCell(frame, args, outName, outCount, opts.xp)
	if showDesc then
		out[#out + 1] = '\n| class="logi-recipes-desc" | ' .. (trim(args.description) or '')
	end

	if foot then
		out[#out + 1] = '\n|}'
	end

	return table.concat(out)
end

function p.grinding(frame)
	return p.build(frame, { label = 'Grinding recipe', xp = true })
end

return p
