-- Module:AlloySmelter — builds Alloy Smelter recipe tables (Name / Ingredients / Alloy Smelter recipe).
-- Clean-room implementation, structured like [[Module:Grinding]] but for a machine that takes two
-- inputs and produces a primary output plus an optional chance byproduct: the Recipe cell reuses
-- our own {{Grid Alloy Smelter Table}} widget. head=1 opens the table + header, foot=1 closes it.

local p = {}

local function trim(s)
	if s == nil then return nil end
	s = mw.text.trim(s)
	if s == '' then return nil end
	return s
end

local function truthy(v)
	v = trim(v)
	return v ~= nil and v ~= '0'
end

-- "Name,Count" -> name, count (also honours an explicit OutputCount)
local function parseOutput(args)
	local out = trim(args.Output) or trim(args.output) or ''
	local count = trim(args.OutputCount) or trim(args.outputcount)
	local n, c = out:match('^(.-)%s*,%s*(%d+)$')
	if n then
		out = n
		count = count or c
	end
	return out, count
end

-- "Name,Chance" -> name, chance (also honours an explicit ByproductChance)
local function parseByproduct(args)
	local bp = trim(args.Byproduct) or trim(args.byproduct)
	if not bp then return nil, nil end
	local chance = trim(args.ByproductChance) or trim(args.byproductchance)
	local n, c = bp:match('^(.-)%s*,%s*([%d%.]+%%?)$')
	if n then
		bp = n
		chance = chance or c
	end
	return bp, chance
end

-- one item, linked if it names a wiki page, else plain text
local function linkItem(v)
	local t = mw.title.new(v)
	return (t and t.exists) and ('[[' .. v .. ']]') or v
end

-- one slot's ingredient text: its ";"-separated alternatives joined with "or", prefixed by a count
local function slotText(key, count)
	if not key then return nil end
	local alts = {}
	for part in key:gmatch('[^;]+') do
		local v = trim(part)
		if v then alts[#alts + 1] = linkItem(v) end
	end
	local text = table.concat(alts, ' or ')
	if count then text = count .. ' ' .. text end
	return text
end

-- Ingredients column: explicit |ingredients=, else the two inputs joined with "+".
local function ingredientsCell(args)
	local ing = trim(args.ingredients)
	if ing then return ing end
	local inputs = {}
	local a = slotText(trim(args.Input) or trim(args.input), trim(args.InputCount) or trim(args.inputcount))
	local b = slotText(trim(args.Input2) or trim(args.input2), trim(args.Input2Count) or trim(args.input2count))
	if a then inputs[#inputs + 1] = a end
	if b then inputs[#inputs + 1] = b end
	local out = {}
	for i, item in ipairs(inputs) do
		out[#out + 1] = '<span class="logi-recipes-ingredient">' .. item .. ((i < #inputs) and ' +' or '') .. '</span>'
	end
	return table.concat(out)
end

-- Recipe column: our alloy-smelter widget (two inputs -> output + optional byproduct).
local function recipeCell(frame, args, outName, outCount)
	local g = { Output = outName }
	if outCount then g.OutputCount = outCount end
	local input = trim(args.Input) or trim(args.input)
	if input then g.Input = input end
	local inputCount = trim(args.InputCount) or trim(args.inputcount)
	if inputCount then g.InputCount = inputCount end
	local input2 = trim(args.Input2) or trim(args.input2)
	if input2 then g.Input2 = input2 end
	local input2Count = trim(args.Input2Count) or trim(args.input2count)
	if input2Count then g.Input2Count = input2Count end
	local bp, bpc = parseByproduct(args)
	if bp then
		g.Byproduct = bp
		if bpc then g.ByproductChance = bpc end
	end
	local grid = frame:expandTemplate{ title = 'Grid Alloy Smelter Table', args = g }
	grid = grid:gsub('<templatestyles.-/>', '')
	return mw.text.trim(grid)
end

function p.alloy(frame)
	local args = frame:getParent().args
	local outName, outCount = parseOutput(args)
	local showName = trim(args.showname) ~= '0'
	local head = truthy(args.head)
	local foot = truthy(args.foot)

	local out = {}

	if head then
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Crafting/styles.css' } }
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Grid Crafting Table/styles.css' } }
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Grid Alloy Smelter Table/styles.css' } }
		out[#out + 1] = '\n{| class="logi-recipes"'
		out[#out + 1] = '\n|- class="logi-recipes-head"'
		if showName then out[#out + 1] = '\n! Name' end
		out[#out + 1] = '\n! Ingredients\n! Alloy Smelter recipe'
	end

	out[#out + 1] = '\n|-'
	if showName then
		local nm = trim(args.name) or ('[[' .. outName .. ']]')
		out[#out + 1] = '\n| class="logi-recipes-name" | ' .. nm
	end
	local ingredients = ingredientsCell(args)
	local notes = trim(args.notes)
	if notes then
		ingredients = ingredients .. '<div class="logi-recipes-note">' .. notes .. '</div>'
	end
	out[#out + 1] = '\n| class="logi-recipes-ingredients" | ' .. ingredients
	out[#out + 1] = '\n| class="logi-recipes-recipe" |\n' .. recipeCell(frame, args, outName, outCount)

	if foot then
		out[#out + 1] = '\n|}'
	end

	return table.concat(out)
end

return p
