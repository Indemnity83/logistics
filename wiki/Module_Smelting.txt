-- Module:Smelting — builds smelting recipe tables (Name / Ingredients / Smelting recipe).
-- Clean-room implementation. Mirrors Module:Crafting: one {{Smelting}} call renders one row;
-- head=1 opens the table + header, foot=1 closes it. The Recipe cell reuses our own
-- {{Grid Smelting Table}} furnace widget.

local p = {}

local function trim(s)
	if s == nil then return nil end
	s = mw.text.trim(s)
	if s == '' then return nil end
	return s
end

local function truthy(v)
	v = trim(v)
	return v ~= nil and v ~= '0'
end

-- "Name,Count" -> name, count (also honours an explicit OutputCount)
local function parseOutput(args)
	local out = trim(args.Output) or trim(args.output) or ''
	local count = trim(args.OutputCount) or trim(args.outputcount)
	local n, c = out:match('^(.-)%s*,%s*(%d+)$')
	if n then
		out = n
		count = count or c
	end
	return out, count
end

-- Ingredients column: explicit |ingredients=, else "<input> + <fuel or Any fuel>".
local function ingredientsCell(args)
	local ing = trim(args.ingredients)
	if ing then return ing end
	local input = trim(args.Input) or trim(args.input) or ''
	local fuel = trim(args.Fuel) or trim(args.fuel) or 'Any fuel'
	local t = input ~= '' and mw.title.new(input)
	if t and t.exists then input = '[[' .. input .. ']]' end
	return input .. ' + ' .. fuel
end

-- Recipe column: our own furnace widget.
local function recipeCell(frame, args, outName, outCount)
	local g = { Output = outName }
	if outCount then g.OutputCount = outCount end
	local input = trim(args.Input) or trim(args.input)
	if input then g.Input = input end
	local fuel = trim(args.Fuel) or trim(args.fuel)
	if fuel then g.Fuel = fuel end
	local xp = trim(args.XP) or trim(args.xp)
	if xp then g.XP = xp end
	-- Styles are loaded once in the table head; strip the per-grid tags so the cell ends on |}.
	local grid = frame:expandTemplate{ title = 'Grid Smelting Table', args = g }
	grid = grid:gsub('<templatestyles.-/>', '')
	return mw.text.trim(grid)
end

function p.smelting(frame)
	local args = frame:getParent().args
	local outName, outCount = parseOutput(args)
	local showName = trim(args.showname) ~= '0'
	local head = truthy(args.head)
	local foot = truthy(args.foot)

	local out = {}

	if head then
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Crafting/styles.css' } }
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Grid Crafting Table/styles.css' } }
		out[#out + 1] = frame:extensionTag{ name = 'templatestyles', args = { src = 'Grid Smelting Table/styles.css' } }
		out[#out + 1] = '\n{| class="logi-recipes"'
		out[#out + 1] = '\n|- class="logi-recipes-head"'
		if showName then out[#out + 1] = '\n! Name' end
		out[#out + 1] = '\n! Ingredients\n! Smelting recipe'
	end

	out[#out + 1] = '\n|-'
	if showName then
		local nm = trim(args.name) or ('[[' .. outName .. ']]')
		out[#out + 1] = '\n| class="logi-recipes-name" | ' .. nm
	end
	out[#out + 1] = '\n| class="logi-recipes-ingredients" | ' .. ingredientsCell(args)
	out[#out + 1] = '\n| class="logi-recipes-recipe" |\n' .. recipeCell(frame, args, outName, outCount)

	if foot then
		out[#out + 1] = '\n|}'
	end

	return table.concat(out)
end

return p
